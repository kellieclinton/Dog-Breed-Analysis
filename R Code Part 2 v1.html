# Explain Outcome.Subtype = AGED, find mean/median of euthanized dogs
filter_type %>% 
  filter(Outcome.Type == "EUTHANIZE") %>% 
  group_by(Outcome.Subtype) %>%
   mutate(
    Intake.Date = mdy(Intake.Date),
    Date.Of.Birth = mdy(Date.Of.Birth)
  ) %>% 
  mutate(Age.Upon.Intake.Date = as.numeric(Intake.Date - Date.Of.Birth), 
         Age_Years = floor(Age.Upon.Intake.Date / 365),
         Age_Days  = Age.Upon.Intake.Date %% 365) %>% 
  summarise(
    median_age_years = floor(median(Age.Upon.Intake.Date, na.rm = TRUE) / 365),
    median_age_days  = median(Age.Upon.Intake.Date, na.rm = TRUE) %% 365,
    mean_age_years   = floor(mean(Age.Upon.Intake.Date, na.rm = TRUE) / 365),
    mean_age_days    = floor(mean(Age.Upon.Intake.Date, na.rm = TRUE) %% 365),
    n = n()
  )
